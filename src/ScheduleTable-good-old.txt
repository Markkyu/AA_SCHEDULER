import React from "react";

// Generate 30-min time slots from 7:00 AM – 8:30 PM
function generateTimeSlots(start = 7, end = 20.5, step = 0.5) {
  const slots = [];
  for (let hour = start; hour <= end; hour += step) {
    const startTimeHour = Math.floor(hour);
    const startTimeMinutes = hour % 1 === 0 ? "00" : "30";
    const period = startTimeHour >= 12 ? "PM" : "AM";
    const formattedHour =
      startTimeHour > 12 ? startTimeHour - 12 : startTimeHour;
    const startTime = `${formattedHour}:${startTimeMinutes} ${period}`;

    slots.push({
      hour,
      startTime,
      disabled: hour >= 12 && hour < 13, // lunch break
    });
  }
  return slots;
}

export default function ScheduleTable({
  headers,
  timeSlotMap,
  onCellClick,
  onCellRightClick,
}) {
  const timeSlots = generateTimeSlots();

  return (
    <table className="min-w-7xl mx-auto my-6 border-collapse">
      <thead>
        <tr>
          <th className="border border-gray-400 px-4 py-2 bg-gray-100">Time</th>
          {headers.map((weekday, i) => (
            <th
              key={i}
              className="border border-gray-400 px-4 py-2 bg-gray-100 font-semibold text-center"
            >
              {weekday}
            </th>
          ))}
        </tr>
      </thead>

      <tbody>
        {timeSlots.map((time, rowIndex) => (
          <tr key={rowIndex} className="bg-white">
            {/* Time Column */}
            <td className="border border-gray-400 px-4 py-2 text-center font-medium">
              {rowIndex % 2 === 0 ? time.startTime : <br />}
            </td>

            {/* Day Columns */}
            {headers.map((day, colIndex) => {
              const startKey = `${day.toUpperCase()}_${time.hour}`;
              const course = timeSlotMap.get(startKey);
              const isDisabled = time.disabled;

              // Skip rendering cell if it’s already covered by a rowspan above
              const isCoveredByRowspan =
                timeSlotMap.get(`${day.toUpperCase()}_${time.hour - 0.5}`) ===
                course;

              if (isCoveredByRowspan && course) return null;

              // Count how many consecutive 30-min blocks this course occupies
              let rowspan = 1;
              if (course) {
                for (let next = time.hour + 0.5; next <= 20.5; next += 0.5) {
                  const nextKey = `${day.toUpperCase()}_${next}`;
                  const nextCourse = timeSlotMap.get(nextKey);
                  if (nextCourse === course) rowspan++;
                  else break;
                }
              }

              // Handle lunch break cell
              if (isDisabled)
                return (
                  <td
                    key={colIndex}
                    className="border border-gray-400 px-4 py-2 text-center bg-gray-100 italic text-gray-500"
                  >
                    Lunch
                  </td>
                );

              // Regular clickable cell
              return (
                <td
                  key={colIndex}
                  rowSpan={course ? rowspan : 1}
                  onClick={() =>
                    !isDisabled &&
                    onCellClick(
                      startKey,
                      `${day.toUpperCase()}_${time.hour + 0.5}`,
                      day,
                      time
                    )
                  }
                  onContextMenu={(e) => {
                    e.preventDefault();
                    if (!isDisabled) onCellRightClick(day, startKey);
                  }}
                  className={`border border-gray-400 px-4 py-2 text-center cursor-pointer font-semibold transition ${
                    course
                      ? "bg-blue-100 hover:bg-blue-200"
                      : "hover:bg-gray-100"
                  }`}
                >
                  {course ? course.course_code : ""}
                </td>
              );
            })}
          </tr>
        ))}
      </tbody>
    </table>
  );
}
